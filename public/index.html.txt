<script>
  let currentUser = null;
  let token = null;
  let allUsers = [];
  let clientsHistory = [];
  const photosCache = {};

  function openSidebar() { document.getElementById('sidebar').classList.add('open'); document.getElementById('sidebar-overlay').classList.add('active'); }
  function closeSidebar() { document.getElementById('sidebar').classList.remove('open'); document.getElementById('sidebar-overlay').classList.remove('active'); }

  async function login() {
    const res = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username: document.getElementById('username').value, password: document.getElementById('password').value })
    });
    const data = await res.json();
    if (res.ok) {
      token = data.token;
      currentUser = data.user;
      initApp();
    } else alert(data.error);
  }

  function logout() { location.reload(); }

  async function initApp() {
    document.getElementById('login-card').classList.add('hidden');
    document.getElementById('app').classList.remove('hidden');
    document.getElementById('settings-btn').classList.remove('hidden');
    document.getElementById('side-username').innerText = currentUser.username;

    await loadUsers();
    if (currentUser.role === 'admin') {
      document.getElementById('admin-wo-card').classList.remove('hidden');
      loadClientsHistory();
    }
    loadWorkOrders();
  }

  async function loadUsers() {
    const res = await fetch('/api/users', { headers: { 'Authorization': 'Bearer ' + token } });
    allUsers = await res.json();
    document.getElementById('wo-lead').innerHTML = allUsers.map(u => `<option value="${u.id}">${u.username}</option>`).join('');
    document.getElementById('wo-helpers').innerHTML = allUsers.map(u => `<option value="${u.username}">${u.username}</option>`).join('');
  }

  async function loadClientsHistory() {
    const res = await fetch('/api/clients-history', { headers: { 'Authorization': 'Bearer ' + token } });
    clientsHistory = await res.json();
    const select = document.getElementById('client-history-select');
    select.innerHTML = '<option value="">-- Sélectionner un client existant --</option>' + 
      clientsHistory.map((c, i) => `<option value="${i}">${c.client_name} (${c.client_address})</option>`).join('');
  }

  function autofillClient() {
    const idx = document.getElementById('client-history-select').value;
    if (idx === '') return;
    const client = clientsHistory[idx];
    document.getElementById('wo-client').value = client.client_name;
    document.getElementById('wo-address').value = client.client_address;
    document.getElementById('wo-nb-hottes').value = client.nb_hottes;
    document.getElementById('wo-nb-portes').value = client.nb_portes;
    document.getElementById('wo-nb-vents').value = client.nb_ventilateurs;
  }

  function switchTab(tab) {
    if (tab === 'active') {
      document.getElementById('tab-active-btn').classList.add('active');
      document.getElementById('tab-history-btn').classList.remove('active');
      document.getElementById('view-active').classList.remove('hidden');
      document.getElementById('view-history').classList.add('hidden');
      loadWorkOrders();
    } else {
      document.getElementById('tab-history-btn').classList.add('active');
      document.getElementById('tab-active-btn').classList.remove('active');
      document.getElementById('view-history').classList.remove('hidden');
      document.getElementById('view-active').classList.add('hidden');
      loadHistoryOrders();
    }
  }

  async function createWorkOrder() {
    const selectedHelpers = Array.from(document.getElementById('wo-helpers').selectedOptions).map(o => o.value);
    
    const payload = {
      client_name: document.getElementById('wo-client').value,
      client_address: document.getElementById('wo-address').value,
      appointment_date: document.getElementById('wo-date').value,
      appointment_time: document.getElementById('wo-time').value,
      team_lead_id: document.getElementById('wo-lead').value,
      helpers: selectedHelpers,
      work_details: document.getElementById('wo-details').value,
      nb_hottes: parseInt(document.getElementById('wo-nb-hottes').value) || 0,
      nb_portes: parseInt(document.getElementById('wo-nb-portes').value) || 0,
      nb_ventilateurs: parseInt(document.getElementById('wo-nb-vents').value) || 0
    };

    await fetch('/api/work-orders', {
      method: 'POST',
      headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });

    alert('Bon de travail créé !');
    if (currentUser.role === 'admin') loadClientsHistory();
    loadWorkOrders();
  }

  async function loadWorkOrders() {
    const res = await fetch('/api/work-orders', { headers: { 'Authorization': 'Bearer ' + token } });
    const data = await res.json();
    renderOrdersList(data, 'wo-list', false);
  }

  async function loadHistoryOrders() {
    const res = await fetch('/api/work-orders?history=true', { headers: { 'Authorization': 'Bearer ' + token } });
    const data = await res.json();
    renderOrdersList(data, 'wo-history-list', true);
  }

  function renderOrdersList(data, containerId, isReadonly) {
    const container = document.getElementById(containerId);
    if (data.length === 0) {
      container.innerHTML = '<div class="card"><p style="color:var(--text-muted); text-align:center;">Aucun bon de travail trouvé.</p></div>';
      return;
    }

    container.innerHTML = data.map(wo => {
      const photos = JSON.parse(wo.photos_data || '{}');
      const timeEntries = JSON.parse(wo.time_entries || '{}');
      const helpersList = JSON.parse(wo.helpers || '[]');
      const isCompleted = wo.status === 'Terminé';

      return `
        <div class="card wo-card ${isCompleted ? 'completed' : ''}">
          <div class="brand-header" style="border:none; padding:0; margin-bottom:15px;">
            <h3 style="margin:0; font-size:1.2rem;">🏢 ${wo.client_name} <span style="color:var(--text-muted); font-size:0.85em;">(#${wo.id})</span></h3>
            <span class="status-badge ${isCompleted ? 'completed' : ''}">${wo.status}</span>
          </div>

          <p><b>📍 Adresse :</b> ${wo.client_address}</p>
          <p><b>📅 Rendez-vous :</b> ${wo.appointment_date} à ${wo.appointment_time}</p>
          <p><b>👨‍✈️ Chef d'équipe :</b> ${wo.team_lead_name} | <b>🧑‍🔧 Helpers :</b> ${helpersList.join(', ') || 'Aucun'}</p>
          <p><b>📝 Travaux demandés :</b> ${wo.work_details}</p>

          <hr style="border-color:var(--card-border); margin:20px 0;">

          <h4 style="margin-top:0; text-transform:uppercase; font-size:0.9rem;">💬 Remarques du Chef d'équipe</h4>
          ${isReadonly 
            ? `<div style="background:var(--input-bg); padding:12px; border-radius:6px; border:1px solid var(--card-border);">${wo.team_lead_comments || '<i>Aucune remarque.</i>'}</div>`
            : `<textarea id="comments-${wo.id}" rows="2" placeholder="Saisir des observations ou remarques sur l'intervention...">${wo.team_lead_comments || ''}</textarea>`
          }

          <hr style="border-color:var(--card-border); margin:20px 0;">

          <h4 style="margin-top:0; text-transform:uppercase; font-size:0.9rem;">⏱️ Heures de Travail Registrées</h4>
          <div class="grid-2">
            <div>
              <label>Chef d'équipe (${wo.team_lead_name})</label>
              ${isReadonly 
                ? `<div>${timeEntries['lead'] || 'Non spécifié'}</div>`
                : `<input type="text" placeholder="ex: 08:00 - 16:30" id="time-lead-${wo.id}" value="${timeEntries['lead'] || ''}">`
              }
            </div>
            ${helpersList.map((h, idx) => `
              <div>
                <label>Helper (${h})</label>
                ${isReadonly 
                  ? `<div>${timeEntries['helper_' + idx] || 'Non spécifié'}</div>`
                  : `<input type="text" placeholder="ex: 08:00 - 16:30" id="time-helper-${wo.id}-${idx}" value="${timeEntries['helper_' + idx] || ''}">`
                }
              </div>
            `).join('')}
          </div>

          <hr style="border-color:var(--card-border); margin:20px 0;">

          <h4 style="margin-top:0; text-transform:uppercase; font-size:0.9rem;">📸 Inspection Photos (Avant / Après)</h4>
          ${renderEquipementPhotos(wo.id, 'Hotte', wo.nb_hottes, photos, isReadonly)}
          ${renderEquipementPhotos(wo.id, 'Porte', wo.nb_portes, photos, isReadonly)}
          ${renderEquipementPhotos(wo.id, 'Ventilateur', wo.nb_ventilateurs, photos, isReadonly)}

          ${!isReadonly ? `
            <div style="margin-top:20px;" class="grid-2">
              <button class="btn-secondary" onclick="saveWorkOrderDetails(${wo.id}, ${helpersList.length})">💾 Enregistrer le rapport</button>
              ${!isCompleted ? `<button class="btn-success" onclick="markCompleted(${wo.id})">✅ Marquer comme Terminé</button>` : ''}
            </div>
          ` : ''}
        </div>
      `;
    }).join('');
  }

  function renderEquipementPhotos(woId, type, count, photos, isReadonly) {
    let html = '';
    for (let i = 1; i <= count; i++) {
      const keyBefore = `${type}_${i}_before`;
      const keyAfter = `${type}_${i}_after`;

      html += `
        <div class="equipement-card">
          <strong style="color:var(--primary-red); font-size:0.95rem;">${type} #${i}</strong>
          <div class="photo-box-grid">
            <div class="photo-box">
              <label>📷 Photo AVANT</label>
              ${!isReadonly ? `<input type="file" accept="image/*" capture="environment" onchange="uploadPhoto(event, ${woId}, '${keyBefore}')">` : ''}
              ${photos[keyBefore] ? `<img src="${photos[keyBefore]}">` : '<p style="font-size:0.8em; color:var(--text-muted); margin: 5px 0 0;">Aucune photo</p>'}
            </div>
            <div class="photo-box">
              <label>📷 Photo APRÈS</label>
              ${!isReadonly ? `<input type="file" accept="image/*" capture="environment" onchange="uploadPhoto(event, ${woId}, '${keyAfter}')">` : ''}
              ${photos[keyAfter] ? `<img src="${photos[keyAfter]}">` : '<p style="font-size:0.8em; color:var(--text-muted); margin: 5px 0 0;">Aucune photo</p>'}
            </div>
          </div>
        </div>
      `;
    }
    return html;
  }

  function uploadPhoto(event, woId, key) {
    const file = event.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = function(e) {
      if (!photosCache[woId]) photosCache[woId] = {};
      photosCache[woId][key] = e.target.result;
      alert('Photo ajoutée ! N\'oubliez pas d\'enregistrer le rapport.');
    };
    reader.readAsDataURL(file);
  }

  async function saveWorkOrderDetails(woId, helpersCount) {
    const timeEntries = {};
    timeEntries['lead'] = document.getElementById(`time-lead-${woId}`).value;

    for (let i = 0; i < helpersCount; i++) {
      const el = document.getElementById(`time-helper-${woId}-${i}`);
      if (el) timeEntries['helper_' + i] = el.value;
    }

    const comments = document.getElementById(`comments-${woId}`).value;

    await fetch('/api/work-orders/' + woId, {
      method: 'PATCH',
      headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        time_entries: timeEntries,
        team_lead_comments: comments,
        photos_data: photosCache[woId] || undefined
      })
    });

    alert('Rapport mis à jour avec succès !');
    loadWorkOrders();
  }

  async function markCompleted(woId) {
    await fetch('/api/work-orders/' + woId, {
      method: 'PATCH',
      headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: 'Terminé' })
    });
    loadWorkOrders();
  }
</script>
